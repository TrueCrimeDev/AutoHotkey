# Crash Logging Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build engine-level crash logging into `bin/AutoHotkey64.exe` so launchers that discard stderr still surface failures with structured records on disk. Three surfaces: `/CrashLog=<path>` flag, `#CrashLog <path>` directive, `/StdErrFile=<path>` tee. Plus exit code 130 for ExternalSignal. SEH `[FATAL]` capture via `SetUnhandledExceptionFilter`. OnError-aware `[ERROR]` records.

**Architecture:** New `source/crashlog.{h,cpp}` module owns the file writer (open-write-flush-close per event, single CRITICAL_SECTION). `source/AutoHotkey.cpp` parses flags, installs the SEH filter and console-control handler, and emits `[START]`/`[EXIT]`. `source/script.cpp` parses the `#CrashLog` directive. `source/error.cpp` adds an OnError-aware `[ERROR]` hook in `ShowError` and a tee in the `PrintErrorStdOut` path. `source/globaldata.{h,cpp}` adds two `LPTSTR` globals.

**Tech Stack:** C++17 (MSVC) / C++20 (mingw). Single new module; rest is small integrations into existing files.

- [ ] **Step 1: Add a private helper to `source/crashlog.cpp`**

```cpp
namespace
{
    // ... existing s_crash_path, s_stderr_path, s_lock, s_lock_initialized ...

    // Append a complete UTF-8 record to the path specified. Open-write-flush-close.
    // Silent no-op if path is null or open fails. NOT reentrant; caller holds s_lock.
    void AppendRaw(LPCTSTR aPath, const char *aBytes, DWORD aLen)
    {
        if (!aPath || aLen == 0) return;
        HANDLE h = CreateFile(aPath,
                              FILE_APPEND_DATA,
                              FILE_SHARE_READ | FILE_SHARE_WRITE,
                              nullptr,
                              OPEN_ALWAYS,
                              FILE_ATTRIBUTE_NORMAL,
                              nullptr);
        if (h == INVALID_HANDLE_VALUE) return;
        DWORD written = 0;
        WriteFile(h, aBytes, aLen, &written, nullptr);
        FlushFileBuffers(h);
        CloseHandle(h);
    }

    // Format a [TAG] line with timestamp + kv pairs. Returns bytes written into aBuf.
    // aBuf must be UTF-8; caller sizes it (4 KB is plenty for one event header).
    DWORD FormatHeader(char *aBuf, size_t aBufSize, const char *aTag, const char *aRest)
    {
        SYSTEMTIME st;
        GetLocalTime(&st);
        return (DWORD)_snprintf_s(aBuf, aBufSize, _TRUNCATE,
            "[%04d-%02d-%02d %02d:%02d:%02d] [%s] %s\n",
            st.wYear, st.wMonth, st.wDay,
            st.wHour, st.wMinute, st.wSecond,
            aTag, aRest);
    }

    // Convert a TCHAR string to a heap-free stack UTF-8 buffer. Returns length.
    // dst_buf is caller-provided; dst_cap is its capacity in bytes.
    int TToUtf8(LPCTSTR aSrc, char *dst_buf, int dst_cap)
    {
        if (!aSrc) { if (dst_cap > 0) dst_buf[0] = 0; return 0; }
        return WideCharToMultiByte(CP_UTF8, 0, aSrc, -1, dst_buf, dst_cap, nullptr, nullptr);
    }
}
```

```cpp
void CrashLog::LogStart(LPCTSTR aScriptPath, LPCTSTR aCmdLine)
{
    EnsureLock();
    if (!s_crash_path) return;
    EnterCriticalSection(&s_lock);

    char header[4096];
    char script_u8[1024] = {};
    char cmdline_u8[2048] = {};
    TToUtf8(aScriptPath, script_u8, sizeof(script_u8));
    TToUtf8(aCmdLine,    cmdline_u8, sizeof(cmdline_u8));

    char rest[3500];
    _snprintf_s(rest, sizeof(rest), _TRUNCATE,
        "pid=%lu ahk=%s script=%s cmdline=\"%s\"",
        GetCurrentProcessId(),
        "2.1-alpha.29+Console",   // pull from RAW_AHK_VERSION instead if a clean path exists
        script_u8, cmdline_u8);

    DWORD n = FormatHeader(header, sizeof(header), "START", rest);
    AppendRaw(s_crash_path, header, n);

    LeaveCriticalSection(&s_lock);
}

void CrashLog::LogExit(int aCode, LPCTSTR aReason)
{
    EnsureLock();
    if (!s_crash_path) return;
    EnterCriticalSection(&s_lock);

    char reason_u8[256] = {};
    TToUtf8(aReason, reason_u8, sizeof(reason_u8));

    char rest[512];
    _snprintf_s(rest, sizeof(rest), _TRUNCATE,
        "pid=%lu code=%d reason=%s",
        GetCurrentProcessId(), aCode, reason_u8);

    char header[1024];
    DWORD n = FormatHeader(header, sizeof(header), "EXIT", rest);
    AppendRaw(s_crash_path, header, n);

    LeaveCriticalSection(&s_lock);
}
```

If `RAW_AHK_VERSION` is reachable via include of `ahkversion.h`, use it instead of the hardcoded version string.

- [ ] **Step 1: Implement `LogError` in `source/crashlog.cpp`**

```cpp
void CrashLog::LogError(LPCTSTR aType, LPCTSTR aMode, LPCTSTR aMessage,
                        LPCTSTR aFile, int aLine, LPCTSTR aWhat, LPCTSTR aExtra,
                        LPCTSTR aStack)
{
    EnsureLock();
    if (!s_crash_path) return;
    EnterCriticalSection(&s_lock);

    char header[4096];
    char type_u8[128] = {}, mode_u8[64] = {};
    char msg_u8[2048] = {}, file_u8[1024] = {};
    char what_u8[256] = {}, extra_u8[2048] = {};
    char stack_u8[8192] = {};
    TToUtf8(aType, type_u8, sizeof(type_u8));
    TToUtf8(aMode, mode_u8, sizeof(mode_u8));
    TToUtf8(aMessage, msg_u8, sizeof(msg_u8));
    TToUtf8(aFile, file_u8, sizeof(file_u8));
    TToUtf8(aWhat, what_u8, sizeof(what_u8));
    TToUtf8(aExtra, extra_u8, sizeof(extra_u8));
    TToUtf8(aStack, stack_u8, sizeof(stack_u8));

    char rest[256];
    _snprintf_s(rest, sizeof(rest), _TRUNCATE,
        "pid=%lu type=%s mode=%s",
        GetCurrentProcessId(), type_u8, mode_u8);

    DWORD n = FormatHeader(header, sizeof(header), "ERROR", rest);
    AppendRaw(s_crash_path, header, n);

    // Continuation lines (indented 2 spaces).
    char body[16384];
    int bn = _snprintf_s(body, sizeof(body), _TRUNCATE,
        "  Message: %s\n"
        "  File: %s\n"
        "  Line: %d\n"
        "  What: %s\n"
        "  Extra: %s\n"
        "  Stack:\n%s",
        msg_u8, file_u8, aLine, what_u8, extra_u8, stack_u8);
    if (bn > 0)
        AppendRaw(s_crash_path, body, (DWORD)bn);

    LeaveCriticalSection(&s_lock);
}
```

```cpp
namespace
{
    LONG WINAPI UnhandledExceptionFilter_Impl(EXCEPTION_POINTERS *pInfo)
    {
        // Defensive: catch any reentrancy via __try, do nothing if we ourselves fault.
        __try
        {
            DWORD code   = pInfo ? pInfo->ExceptionRecord->ExceptionCode    : 0;
            PVOID addr   = pInfo ? pInfo->ExceptionRecord->ExceptionAddress : nullptr;

            // Best-effort context from the script. Globals may be null/stale.
            LPCTSTR last_file = (g_script.mCurrFileIndex >= 0 && Line::sSourceFile)
                                ? Line::sSourceFile[g_script.mCurrFileIndex]
                                : _T("");
            int     last_line = g_script.mCurrLine ? g_script.mCurrLine->mLineNumber : 0;
            LPCTSTR last_hk   = g_script.mThisHotkeyName ? g_script.mThisHotkeyName : _T("");

            CrashLog::LogFatal(code, addr, last_file, last_line, last_hk);
            CrashLog::LogExit(11, _T("Fatal"));
        }
        __except (EXCEPTION_EXECUTE_HANDLER)
        {
            // Swallow — we tried our best.
        }
        return EXCEPTION_CONTINUE_SEARCH;
    }
}

void CrashLog::InstallExceptionFilter()
{
    SetUnhandledExceptionFilter(UnhandledExceptionFilter_Impl);
}
```
